## 03 — src/catalogue.js — 🔴 Critical Bug

**Problem:** `applyCatalogueItem` spreads the old draft item, then only replaces CP when the selected catalogue record contains CP. Selecting NEW ITEM with no CP retains OLD ITEM's CP. Reproduction retained `cost: '50'` and calculated profit as if that belonged to the new material.

**Solution:** Always assign the selected item's CP, including blank when it is unavailable. Replace `applyCatalogueItem` with the function below. Finding 04 extends the unit/rate helpers this function uses.

```js
export function applyCatalogueItem(item, record, type) {
  const unit = catalogueUnit(record, type);
  const rate = catalogueRate(record, type);
  const changed = unit !== (item.billingUnit || 'nos');
  const cost = Number(record.costMinor);
  const hasCost = type === 'sale' && record.costMinor !== '' &&
    record.costMinor != null && record.costBillingUnit === unit &&
    Number.isSafeInteger(cost) && cost >= 0;
  return {
    ...item,
    description: record.name,
    billingUnit: unit,
    rate: Number.isSafeInteger(rate) ? String(rate / 100) : '',
    cost: hasCost ? String(cost / 100) : '',
    ...(changed ? {
      grouped: false,
      measurementMode: ['sqft', 'rft'].includes(unit) ? 'dimensions' : 'quantity',
      measurements: ['sqft', 'rft'].includes(unit) ? [blankMeasurement()] : []
    } : {})
  };
}
```

Blank CP means profit stays pending until entered; it must not silently become zero. This changes defaults only, not CP on previously issued invoices.

## 05 — src/useStock.js — 🟡 Optimization/Redundancy

**Problem:** Stored stock queue validation checks only ID/action. A payload with `reviewMode: 'invoice'` but no movements passes and crashes rendering at `p.movements.entries()`. The original data remains stored, but the Stock UI cannot open to repair it.

**Solution:** Validate structural fields before projecting queued work. Replace the nested `read()` function with this version. It preserves unreadable storage and lets the existing invalid-queue UI report recovery.

```js
function read(storageKey) {
  const value = JSON.parse(localStorage.getItem(storageKey) || 'null');
  const rows = value?.queue || (value ? [value] : []);
  if (!Array.isArray(rows)) throw new Error('The saved stock queue cannot be read.');
  const actions = new Set(['importStock','createStockItem','updateStockItem',
    'bulkUpdateStockItems','deleteStockItems','recordStock','reviewInvoiceStock','reverseStock']);
  const ids = new Set();
  for (const row of rows) {
    const payload = row?.payload;
    if (!actions.has(row?.action) || typeof payload?.id !== 'string' ||
        !payload.id || ids.has(payload.id)) {
      throw new Error('The saved stock queue is invalid. Export recovery data before repairing it.');
    }
    ids.add(payload.id);
    const projected = row.action === 'recordStock' && Array.isArray(payload.movements) ||
      row.action === 'reviewInvoiceStock' && payload.reviewMode === 'invoice';
    if (projected) {
      if (!Array.isArray(payload.movements) || payload.movements.length > 100 ||
          payload.reviewItemIds !== undefined && !Array.isArray(payload.reviewItemIds) ||
          payload.priceUpdates !== undefined && !Array.isArray(payload.priceUpdates)) {
        throw new Error('Saved stock movement details are invalid. Export recovery data.');
      }
      for (const movement of payload.movements) {
        if (typeof movement?.stockItemId !== 'string' || typeof movement.unit !== 'string') {
          throw new Error('A saved stock material is invalid. Export recovery data.');
        }
        stockDecimal(movement.quantity, { signed: true });
      }
    }
  }
  return rows.map(row => row.rejected ? { ...row, failed: true } : row);
}
```

Import `stockDecimal` from `stock.js`. Do not automatically delete or overwrite malformed queues. This catches the reproduced structural crash; it is not a substitute for backend business validation.

## 09 — src/AccountDialog.jsx — 🟡 Optimization/Redundancy

**Problem:** Each popup captures and restores body overflow independently. Removing one dialog while another stays mounted can restore scrolling underneath the remaining modal. This was reproduced with two open dialogs and removal of the first. Listeners themselves are cleaned up; the problem is overlapping ownership of shared state.

**Solution:** Use a reference-counted scroll lock and release it exactly once per mounted dialog. Replace the component with this version.

```jsx
import React, { useEffect, useId, useRef } from 'react';
import { createPortal } from 'react-dom';

let scrollLocks = 0;
let originalOverflow = '';
function lockBodyScroll() {
  if (scrollLocks++ === 0) {
    originalOverflow = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
  }
  let released = false;
  return () => {
    if (released) return;
    released = true;
    if (--scrollLocks === 0) document.body.style.overflow = originalOverflow;
  };
}

export default function AccountDialog({ title, onClose, busy = false,
  className = '', initialFocus = null, onKeyDown, children }) {
  const dialog = useRef(null);
  const titleId = useId();
  useEffect(() => {
    const element = dialog.current;
    const release = lockBodyScroll();
    try { if (!element.open) element.showModal(); }
    catch (error) { release(); throw error; }
    return () => { element.close(); release(); };
  }, []);
  useEffect(() => {
    if (initialFocus) dialog.current?.querySelector(initialFocus)?.focus();
  }, [initialFocus]);
  return createPortal(
    <dialog ref={dialog} className={`account-dialog ${className}`}
      aria-labelledby={titleId} onKeyDown={onKeyDown}
      onCancel={event => { event.preventDefault(); if (!busy) onClose(); }}>
      <div className="account-dialog-heading">
        <h2 id={titleId}>{title}</h2>
        <button type="button" className="outline" disabled={busy} onClick={onClose}
          aria-label="Close popup" title="Close popup (Escape)">Close</button>
      </div>
      {children}
    </dialog>, document.body
  );
}
```

## 13 — google-apps-script/Code.gs — 🟢 Best Practice

**Problem:** Payment writes repeat a different literal-string guard from account writes. Account writes handle leading whitespace; payment writes only check the first character. Update paths also pre-escape a value and then map the whole row again. This is inconsistent security hardening and duplicated logic, not a demonstrated live exploit.

**Solution:** Use one literal-cell helper and escape once at the write boundary. Google documents that strings beginning with `=` are interpreted as formulas by setValues; plain-text formatting alone should not replace the explicit guard. [Google Range.setValues documentation](https://developers.google.com/apps-script/reference/spreadsheet/range#setValues(Object)).

```js
function literalCell_(value) {
  return typeof value === 'string' && /^[\t\r\n]|^\s*[=+\-@]/.test(value)
    ? "'" + value : value;
}
function literalAccountRow_(headers, value, previous = []) {
  return headers.map((key, index) => literalCell_(
    Object.prototype.hasOwnProperty.call(value, key)
      ? value[key] ?? '' : previous[index] ?? ''
  ));
}
```

For payment restore/update/delete writes, replace the existing `.map(value => ...)` escaping expressions with `values.map(literalCell_)`. For payment creation use `row.map(literalCell_)`. In the update `fields.forEach` assignment, set `values[headers.indexOf(key)] = changed[key]` without pre-escaping; escape only in `setValues`.

Do not log request bodies or tokens when diagnosing authorization failures. The existing performance logs avoid doing this.

## 17 — public/sw.js — 🟡 Optimization/Redundancy

**Problem:** Script/style fallback only handles rejected fetches. An HTTP 404/503 resolves successfully, so the worker returns the failed response even if that exact hashed asset is cached. This is a static-review deployment/recovery gap; the existing tests cover offline network rejection, not HTTP failures.

**Solution:** Replace the script/style branch's respondWith call with this block. It only falls back for transient/server or missing-asset responses and preserves successful network responses.

```js
event.respondWith((async () => {
  try {
    const response = await fetch(request);
    if (response.ok) {
      remember(event, request, response);
      return response;
    }
    if (response.status === 404 || response.status >= 500) {
      return await cached(request) || response;
    }
    return response;
  } catch {
    return await cached(request) || Response.error();
  }
})());
```

Do not cache Sheets responses or credentials. Production offline availability of a lazy page still depends on its chunk having been fetched previously; this change does not claim otherwise.

## 18 — src/StockUpdateWorkspace.jsx — 🟡 Optimization/Redundancy

**Problem:** The stock search combobox has arrow/Enter selection but no Escape dismissal, and its suggestions remain open whenever query text matches even after moving focus elsewhere. Escape currently reaches the dialog's cancel handler, potentially closing the whole workspace.

**Solution:** Extract the search control below. It supplies semantic options, explicit popup state, stable IDs, keyboard dismissal and a shortcut annotation. ```jsx
import React, { useId, useState } from 'react';

export default function StockSearch({ inputRef, query, onQuery, matches, onSelect, disabled }) {
  const id = useId();
  const [open, setOpen] = useState(false);
  const [active, setActive] = useState(0);
  const expanded = open && matches.length > 0;
  const index = Math.min(active, Math.max(0, matches.length - 1));
  const choose = item => { setOpen(false); setActive(0); onSelect(item); };
  return <div className="stock-pos-search" onBlur={event => {
    if (!event.currentTarget.contains(event.relatedTarget)) setOpen(false);
  }}>
    <input id="stock-pos-search" ref={inputRef} type="search" autoComplete="off"
      role="combobox" aria-label="Search stock item" aria-autocomplete="list"
      aria-expanded={expanded} aria-controls={expanded ? id : undefined}
      aria-activedescendant={expanded ? `${id}-${index}` : undefined}
      data-hotkey="mod+k" aria-keyshortcuts="Control+K Meta+K"
      placeholder="Search item…" disabled={disabled} value={query}
      onFocus={() => setOpen(Boolean(query.trim()))}
      onChange={event => { onQuery(event.target.value); setActive(0); setOpen(true); }}
      onKeyDown={event => {
        if (event.isComposing || event.repeat || event.ctrlKey || event.metaKey || event.altKey) return;
        if (event.key === 'Escape' && expanded) {
          event.preventDefault(); event.stopPropagation(); setOpen(false);
        } else if (['ArrowDown', 'ArrowUp'].includes(event.key) && matches.length) {
          event.preventDefault(); setOpen(true);
          setActive(previous => !expanded ? 0 : Math.max(0, Math.min(matches.length - 1,
            previous + (event.key === 'ArrowDown' ? 1 : -1))));
        } else if (event.key === 'Enter') {
          event.preventDefault();
          if (expanded && matches[index]) choose(matches[index]);
        }
      }}/>
    <kbd className="stock-key-hint" aria-hidden="true">Ctrl/⌘ K</kbd>
    {expanded && <div id={id} role="listbox" aria-label="Matching stock items"
      className="stock-pos-suggestions">
      {matches.map((item, option) => <button type="button" role="option"
        id={`${id}-${option}`} key={item.id} tabIndex={-1}
        aria-selected={index === option} onMouseDown={event => event.preventDefault()}
        onClick={() => choose(item)}>
        <strong>{item.name}{item.code ? ` · ${item.code}` : ''}</strong>
        <span>{item.balance} {item.baseUnit === 'UNIT' ? 'CSV unit' : item.baseUnit}</span>
      </button>)}
    </div>}
  </div>;
}

import { minor as parseMoneyMinor } from './invoiceAmounts.js';

function parseMinor(value, label, allowZero = false) {
